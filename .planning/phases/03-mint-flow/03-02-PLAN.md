Hook implementation:
```
export function useMintEgg() {
  const { writeContractAsync, data: txHash, isPending, error: writeError, reset } = useWriteContract();
  const { isLoading: isConfirming, isSuccess, isError: isConfirmError, error: confirmError } = useWaitForTransactionReceipt({ hash: txHash });

  const mint = async (tokenURI: string) => {
    await writeContractAsync({
      address: EASTER_EGG_ADDRESS,
      abi: EASTER_EGG_ABI,
      functionName: "mint",
      args: [tokenURI],
      chainId: APP_CHAIN_ID,
    });
  };

  return {
    mint,
    isPending,      // tx submitted to wallet, waiting for user signature
    isConfirming,   // tx sent, waiting for on-chain confirmation
    isSuccess,      // tx confirmed on-chain
    isError: !!writeError || isConfirmError,
    error: writeError || confirmError,
    txHash,
    reset,          // reset state for retry
  };
}
```

This follows the exact pattern from main-apostole's useMint.ts but adapted for EasterEgg's `mint(string tokenURI)` signature. Setting `chainId` in the writeContract call ensures wagmi prompts chain switch if user is on wrong network (addresses Pitfall 6 from research).
